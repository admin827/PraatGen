# Electroglottogram support procedures: the mandatory cycle guard that
# blocks the To TextGrid (closed glottis) / To AmplitudeTier (levels)
# segfault, and the spectral-threshold de-noiser used to rescue
# low-SNR EGG recordings for contact-quotient measurement.
#
# This file lives at: egg/eml-egg-procedures.praat
# Include from wrapper scripts via: include ../egg/eml-egg-procedures.praat
#
# Dependencies: none (uses only core Praat commands)
#
# Method, thresholds, and provenance: see
#   BEST_PRACTICES_EGG_CONTACT_QUOTIENT.md
# Command syntax, arity, and failure modes: see
#   COMMANDS_Electroglottogram.txt
#
# Procedure index:
#   @emlEggCycleGuard        — segfault guard; sets .safe (1 = analyzable)
#   @emlEggSpectralThreshold — spectral-threshold de-noiser; returns .resultId

# ----------------------------------------------------------------------------
# @emlEggCycleGuard: .eggId, .floor, .ceiling
#
# MANDATORY before To TextGrid (closed glottis) or To AmplitudeTier (levels).
# Those two commands segfault Praat — no catchable error — when no cycle falls
# in [.floor, .ceiling] (silent channel, noise-only channel, or floor >=
# ceiling). Sets .safe = 1 when at least one voiced cycle is present and the
# mean F0 is defined and >= .floor; .safe = 0 otherwise.
#
# The nested ifs are required. Praat's "and"/"or" do not short-circuit, so a
# single compound condition would evaluate .f0 < .floor even when .f0 is
# undefined.
# ----------------------------------------------------------------------------
procedure emlEggCycleGuard: .eggId, .floor, .ceiling
    selectObject: .eggId
    .snd = To Sound
    selectObject: .snd
    .p = noprogress To Pitch (raw cross-correlation): 0.0, .floor,
        ... .ceiling, 15, "no", 0.03, 0.45, 0.01, 0.35, 0.14
    selectObject: .p
    .voiced = Count voiced frames
    .f0 = Get mean: 0, 0, "Hertz"
    removeObject: .p, .snd
    .safe = 1
    if .voiced = 0
        .safe = 0
    endif
    if .safe = 1
        if .f0 = undefined
            .safe = 0
        endif
    endif
    if .safe = 1
        if .f0 < .floor
            .safe = 0
        endif
    endif
endproc

# ----------------------------------------------------------------------------
# @emlEggSpectralThreshold: .soundId, .thresholdBelowPeak, .lowPassHz
#
# 4:1 downward dB expansion below threshold, phase preserved. Expansion rather
# than zeroing avoids transients as components cross the threshold. Returns
# .resultId (a Sound). Offer as a rescue for a recording that is otherwise
# unusable at low SNR, and TELL THE USER — it changes their data.
#
# Two traps (both authoritative in COMMANDS_Spectrum.txt, repeated where they
# bite): use To Spectrum: "no" (padding inflates To Sound length); anchor the
# threshold to raw Spectrum magnitude, never an Ltas peak (~91 dB mismatch).
# ----------------------------------------------------------------------------
procedure emlEggSpectralThreshold: .soundId, .thresholdBelowPeak, .lowPassHz
    selectObject: .soundId
    .spec = To Spectrum: "no"
    selectObject: .spec
    .magCopy = Copy: "mag"
    selectObject: .magCopy
    Formula: ~ sqrt (self [1, col] ^ 2 + self [2, col] ^ 2)
    .magMat = To Matrix
    selectObject: .magMat
    .rawPeak = Get maximum
    removeObject: .magCopy, .magMat
    .thrLin = .rawPeak * 10 ^ (- .thresholdBelowPeak / 20)
    selectObject: .spec
    Formula: ~ self * (if sqrt (self [1, col] ^ 2 + self [2, col] ^ 2)
        ... < .thrLin then ((sqrt (self [1, col] ^ 2 + self [2, col] ^ 2)
        ... + 1e-30) / .thrLin) ^ 3 else 1 fi)
    selectObject: .spec
    .denoised = To Sound
    removeObject: .spec
    if .lowPassHz > 0
        selectObject: .denoised
        .resultId = Filter (pass Hann band): 0, .lowPassHz, 100
        removeObject: .denoised
    else
        .resultId = .denoised
    endif
endproc
